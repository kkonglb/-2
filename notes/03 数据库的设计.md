<h3 id="vrIOE">商品信息表</h3>
```sql
CREATE TABLE `product` (
  `id` int(11) NOT NULL AUTO_INCREMENT COMMENT '商品的ID，主键自增',
  `name` varchar(255) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '商品名',
  `detail` longtext COLLATE utf8mb4_bin COMMENT '商品的简介',
  `cover_list` longtext COLLATE utf8mb4_bin COMMENT '商品封面的列表',
  `old_level` int(2) DEFAULT NULL COMMENT '新旧程度',
  `category_id` int(11) DEFAULT NULL COMMENT '所属商品分类的ID',
  `user_id` int(11) DEFAULT NULL COMMENT '发布者用户ID',
  `inventory` int(11) DEFAULT NULL COMMENT '库存',
  `price` decimal(10,2) DEFAULT NULL COMMENT '价格',
  `is_bargain` tinyint(1) DEFAULT NULL COMMENT '是否支持砍价',
  `create_time` datetime DEFAULT NULL COMMENT '创建时间',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_bin COMMENT='商品信息表';
```

<h3 id="btHSb">商品类别表</h3>
```sql
CREATE TABLE `category` (
  `id` int(11) NOT NULL AUTO_INCREMENT COMMENT '商品类别ID',
  `name` varchar(255) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '类别名',
  `is_use` tinyint(1) DEFAULT NULL COMMENT '是否启用',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_bin COMMENT='商品类别表';
```

<h3 id="jqxM7">互动行为信息表</h3>
```sql
CREATE TABLE `interaction` (
  `id` int(11) NOT NULL AUTO_INCREMENT COMMENT '互动行为主键ID',
  `user_id` int(11) DEFAULT NULL COMMENT '用户ID',
  `product_id` int(11) DEFAULT NULL COMMENT '商品ID',
  `type` int(2) DEFAULT NULL COMMENT '行为类型（1：收藏；2：浏览；3： 想要）',
  `create_time` datetime DEFAULT NULL COMMENT '行为互动时间',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_bin COMMENT='互动行为信息表';
```

<h3 id="nrUD0">订单信息表</h3>
```sql
CREATE TABLE `orders` (
  `id` int(11) NOT NULL AUTO_INCREMENT COMMENT '订单主键',
  `code` varchar(255) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '订单号',
  `detail` varchar(255) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '备注',
  `product_id` int(11) DEFAULT NULL COMMENT '商品ID',
  `buy_price` decimal(10,2) DEFAULT NULL COMMENT '购买时的价格',
  `trade_status` tinyint(1) DEFAULT NULL COMMENT '交易状态',
  `trade_time` datetime DEFAULT NULL COMMENT '交易时间',
  `create_time` datetime DEFAULT NULL COMMENT '创建时间',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_bin COMMENT='订单信息表';
```

<h3 id="fe0nI">消息表</h3>
```sql
CREATE TABLE `message` (
  `id` int(11) NOT NULL AUTO_INCREMENT COMMENT '消息表主键ID',
  `user_id` int(11) DEFAULT NULL COMMENT '接收者用户ID',
  `content` varchar(255) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '消息体',
  `is_read` tinyint(1) DEFAULT NULL COMMENT '是否已经阅读',
  `create_time` datetime DEFAULT NULL COMMENT '发送时间',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_bin COMMENT='消息表';
```

<h3 id="jzQE6">操作日志表</h3>

```sql
CREATE TABLE `operation_log` (

  `user_id` int(11) DEFAULT NULL COMMENT '用户ID',
  `detail` varchar(255) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '描述',
  `create_time` datetime DEFAULT NULL COMMENT '创建时间',
  PRIMARY KEY (`id`)

```
